function MenuItem({
  icon,
  text,
  active = false,
}: {
  icon: React.ReactNode;
  text: string;
  active?: boolean;
}) {
  return (
    <div
      className={`flex items-center gap-3 p-4 rounded-xl cursor-pointer
      ${
        active
          ? "bg-green-500 text-white"
          : "hover:bg-green-50"
      }`}
    >
      {icon}
      <span>{text}</span>
    </div>
  );
}

function StatCard({
  title,
  value,
  color,
  icon,
}: {
  title: string;
  value: string;
  color: string;
  icon: string;
}) {
  return (
    <div className={`${color} p-5 rounded-3xl shadow-sm`}>
      <div className="text-3xl mb-2">{icon}</div>
      <p className="text-gray-600">{title}</p>
      <h3 className="text-3xl font-bold">{value}</h3>
    </div>
  );
}

function AIBox({
  title,
  detail,
}: {
  title: string;
  detail: string;
}) {

      <h4 className="font-semibold">{title}</h4>
      <p className="text-gray-600 text-sm">{detail}</p>
    </div>
  );
}
